  # ENet 1.3.18's decoder seed promotes a byte to signed int before <<24.
  # Build a corrected copy, preserving the verified dependency source and wire bits.
  file(READ "${IRON_DEPS_DIR}/enet/compress.c" iron_enet_codec)
  string(REPLACE "decodeCode |= * inData ++ << 24;" "decodeCode |= (enet_uint32) * inData ++ << 24;" iron_enet_safe_codec "${iron_enet_codec}")
  file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/deps/enet-compress.c" CONTENT "${iron_enet_safe_codec}" @ONLY)
  get_target_property(iron_enet_sources enet SOURCES)
  set(iron_enet_safe_sources)
  foreach(source IN LISTS iron_enet_sources)
    if(source STREQUAL "compress.c" OR source STREQUAL "${IRON_DEPS_DIR}/enet/compress.c")
      list(APPEND iron_enet_safe_sources "${CMAKE_BINARY_DIR}/deps/enet-compress.c")
    elseif(IS_ABSOLUTE "${source}")
      list(APPEND iron_enet_safe_sources "${source}")
    else()
      list(APPEND iron_enet_safe_sources "${IRON_DEPS_DIR}/enet/${source}")
    endif()
  endforeach()
  set_property(TARGET enet PROPERTY SOURCES ${iron_enet_safe_sources})
  set(ENET_INCLUDE_DIR "${IRON_DEPS_DIR}/enet/include")
  set(ENET_LIBRARY enet)
